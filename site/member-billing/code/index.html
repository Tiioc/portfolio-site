The Apps Script: a Front desk menu and the 8am fix list email.

/**
 * Member Billing Tracker: Apps Script.
 * All the checks are formulas in the sheet. This script only adds a menu and emails the
 * Fix List to the front desk each morning.
 *
 * Setup: in the Google Sheet, open Extensions > Apps Script, paste this file, save,
 * reload the sheet, then use the "Front desk" menu.
 */

function onOpen() {
  SpreadsheetApp.getUi()
    .createMenu('Front desk')
    .addItem("Email me today's fix list", 'emailFixList')
    .addItem('Send the fix list every morning', 'turnOnDailyEmail')
    .addToUi();
}

/** Reads the Fix List tab and returns one object per problem. */
function readFixList(sheet) {
  const values = sheet.getRange(5, 1, sheet.getLastRow() - 4, 7).getValues();
  return values
    .filter(r => r[1] !== '')
    .map(r => ({ memberId: r[1], name: r[2], phone: r[3], problem: r[4], fix: r[5], ref: r[6] }));
}

/** Builds the plain email text. Kept separate so it can be tested outside Google. */
function buildFixListEmail(items, dateText) {
  if (items.length === 0) {
    return { subject: 'Fix list for ' + dateText + ': nothing to fix', body: 'Nothing on the fix list today.' };
  }
  const lines = items.map((it, i) =>
    (i + 1) + '. ' + it.name + ' (' + it.memberId + ', ' + it.phone + '): ' + it.problem + '. ' + it.fix + '. [' + it.ref + ']');
  return {
    subject: 'Fix list for ' + dateText + ': ' + items.length + (items.length === 1 ? ' thing' : ' things') + ' to fix',
    body: 'Good morning! Here is today\'s billing fix list.\n\n' + lines.join('\n') +
      '\n\nAlways look members up by member ID, not by name. Two members can have the same name.'
  };
}

function emailFixList() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const items = readFixList(ss.getSheetByName('Fix List'));
  const dateText = Utilities.formatDate(new Date(), ss.getSpreadsheetTimeZone(), 'MMM d, yyyy');
  const msg = buildFixListEmail(items, dateText);
  MailApp.sendEmail(Session.getActiveUser().getEmail(), msg.subject, msg.body);
}

/** Creates one daily trigger at about 8am, replacing any earlier one. */
function turnOnDailyEmail() {
  ScriptApp.getProjectTriggers()
    .filter(t => t.getHandlerFunction() === 'emailFixList')
    .forEach(t => ScriptApp.deleteTrigger(t));
  ScriptApp.newTrigger('emailFixList').timeBased().everyDays(1).atHour(8).create();
  SpreadsheetApp.getUi().alert('Done. The fix list will be emailed to you every morning around 8am.');
}

if (typeof module !== 'undefined') module.exports = { readFixList, buildFixListEmail, emailFixList, turnOnDailyEmail };

"""Tests for the Member Billing Tracker.

  python3 test_tracker.py                  # answer key vs a separate Python version of the rules
  python3 test_tracker.py results.xlsx     # also check a spreadsheet engine's computed values
                                           # (LibreOffice recalc, or a Google Sheets export)
"""
import json, sys, collections
from openpyxl import load_workbook

data = json.load(open("/home/user/finances/member-billing/answer_key.json"))
KEY, MEM, PAY, REF = data["answer_key"], data["members"], data["payments"], data["refunds"]

def check(name, ok, detail=""):
    print(("PASS " if ok else "FAIL ") + name + (f"  ({detail})" if detail and not ok else ""))
    if not ok: fails.append(name)

# ---- A separate version of the rules, written in Python -----------------------------------
def rules(by):
    """by='id' looks members up by ID (the real rule). by='name' looks them up by name, first match."""
    first_by_name = {}
    for m in MEM.values(): first_by_name.setdefault(m["name"], m)
    who = (lambda p: MEM[p["mid"]]) if by == "id" else (lambda p: first_by_name[MEM[p["mid"]]["name"]])
    same = (lambda a, b: a["mid"] == b["mid"]) if by == "id" else \
           (lambda a, b: MEM[a["mid"]]["name"] == MEM[b["mid"]]["name"])
    out, seen = {}, collections.Counter()
    for p in PAY:
        m = who(p); dues_paid = p["type"] == "Dues" and p["result"] == "Paid"
        if dues_paid and p["card"] == m["card"]:
            seen[(m["id"], p["period"])] += 1
        if p["disputed"] == "Yes": out[p["txn"]] = "Disputed charge"
        elif dues_paid and p["card"] != m["card"]: out[p["txn"]] = "Paid with another member's card"
        elif dues_paid and m["status"] != "Active" and p["date"] >= m["since"]:
            out[p["txn"]] = "Charged after " + ("pausing" if m["status"] == "Paused" else "cancelling")
        elif dues_paid and p["card"] == m["card"] and seen[(m["id"], p["period"])] > 1:
            out[p["txn"]] = "Charged twice this month"
        elif p["type"] == "Dues" and p["result"] == "Failed" and not any(
                same(q, p) and q["period"] == p["period"] and q["type"] == "Dues" and q["result"] == "Paid" for q in PAY):
            out[p["txn"]] = "Card failed, not retried"
    for q in REF:
        sent = any(same(p, q) and p["type"] == "Refund" and p["amount"] == q["amount"] and p["date"] >= q["date"] for p in PAY)
        if not sent: out[q["req"]] = "Refund approved but never sent"
    return out

def compare(label, got):
    caught = {k for k in KEY if got.get(k) == KEY[k]}
    wrong_label = {k for k in KEY if k in got and got[k] != KEY[k]}
    missed = set(KEY) - set(got)
    extra = set(got) - set(KEY)
    print(f"  {label}: caught {len(caught)} of {len(KEY)} planted, missed {len(missed)}, "
          f"wrong label {len(wrong_label)}, false flags {len(extra)}")
    return caught, missed, wrong_label, extra

def same_name_tests(label, got):
    a = next(p["txn"] for p in PAY if p["mid"] == "M-1012" and p["period"] == "2026-10" and p["type"] == "Dues")
    b = next(p["txn"] for p in PAY if p["mid"] == "M-1031" and p["period"] == "2026-10" and p["type"] == "Dues")
    check(f"{label} same-name 1: active Chris Morgan (M-1012) October charge is not flagged", a not in got)
    check(f"{label} same-name 2: paused Chris Morgan (M-1031) October charge is flagged", got.get(b) == "Charged after pausing")
    check(f"{label} same-name 3: M-1031's refund is still owed even though M-1012 got a $95 refund",
          got.get("R-04") == "Refund approved but never sent")
    check(f"{label} same-name 4: M-1012's refund counts as sent", "R-02" not in got)
    m31 = [p["txn"] for p in PAY if p["mid"] == "M-1031" and p["period"] != "2026-10"]
    check(f"{label} same-name 5: M-1031's earlier charges are not flagged as another member's card",
          all(t not in got for t in m31))

print("1. Answer key vs the Python version of the rules (lookups by member ID)")
by_id = rules("id")
c, mi, wl, ex = compare("by ID", by_id)
check("python: every planted problem caught with the right label", len(c) == len(KEY))
check("python: no false flags", not ex, ex)
same_name_tests("python", by_id)
