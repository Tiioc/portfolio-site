The engine: staffed-hours math, the health check, the ownership queue, reminders, the weekly scorecard, sorting and scoring.

/**
 * InboxOwner core: response-time audit, ownership/SLA status and rules triage
 * for a shared inbox.
 *
 * Plain JavaScript with no dependencies, so the same file runs in the browser
 * demo, in Node tests and in Google Apps Script (where it reads a real Gmail
 * inbox).
 *
 * Time convention: every timestamp is "studio wall-clock milliseconds", i.e.
 * Date.UTC(year, month, day, hour, minute) of the local time at the business.
 * Callers convert real dates once at the edge; nothing in here depends on the
 * machine's time zone.
 */
(function (root) {
  'use strict';

  const HOUR = 3600000;
  const MIN = 60000;
  const DAY = 24 * HOUR;

  /* ------------------------------------------------------------ helpers */

  function median(xs) { return quantile(xs, 0.5); }

  // Linear-interpolated quantile (same method as Google Sheets PERCENTILE).
  function quantile(xs, q) {
    if (!xs.length) return null;
    const s = xs.slice().sort((a, b) => a - b);
    const pos = (s.length - 1) * q;
    const lo = Math.floor(pos), hi = Math.ceil(pos);
    return s[lo] + (s[hi] - s[lo]) * (pos - lo);
  }

  function round(x, d) { const f = Math.pow(10, d || 0); return Math.round(x * f) / f; }

  /* ----------------------------------------------------- business hours */

  /**
   * hours: {0..6: [openHour, closeHour] | null}, 0 = Sunday.
   * Returns business minutes between two wall-clock timestamps.
   */
  function businessMinutes(start, end, hours) {
    if (end <= start) return 0;
    let total = 0;
    let dayStart = Math.floor(start / DAY) * DAY;
    while (dayStart < end) {
      const h = hours[new Date(dayStart).getUTCDay()];
      if (h) {
        const open = dayStart + h[0] * HOUR, close = dayStart + h[1] * HOUR;
        const a = Math.max(open, start), b = Math.min(close, end);
        if (b > a) total += (b - a) / MIN;
      }
      dayStart += DAY;
    }
    return total;
  }

  /** Adds business minutes to a wall-clock timestamp (used for SLA due times). */
  function addBusinessMinutes(start, minutes, hours) {
    let t = start, left = minutes;
    for (let guard = 0; guard < 400; guard++) {
      const dayStart = Math.floor(t / DAY) * DAY;
      const h = hours[new Date(dayStart).getUTCDay()];
      if (h) {
        const open = dayStart + h[0] * HOUR, close = dayStart + h[1] * HOUR;
        const from = Math.max(t, open);
        if (from < close) {
          const avail = (close - from) / MIN;
          if (avail >= left) return from + left * MIN;
          left -= avail;
        }
      }
      t = dayStart + DAY;
    }
    throw new Error('addBusinessMinutes: no business hours configured');
  }

  /* ------------------------------------------------------------- audit */

  /**
   * Per-thread facts. A thread is {id, category?, messages:[{from:'customer'|'staff', staff?, at}]}
   * sorted by time. Only metadata is used; message bodies are never needed.
   */
  function analyzeThread(thread, cfg) {
    const msgs = thread.messages;
    const firstIn = msgs.find(m => m.from === 'customer');
    if (!firstIn) return null; // outbound-only thread: not a customer request
    const firstReply = msgs.find(m => m.from === 'staff' && m.at > firstIn.at);
    const replyAt = firstReply ? firstReply.at : null;

    // Duplicate reply: two or more different staff answer the same customer
    // message before the customer writes again.
    let duplicate = false;
    for (let i = 0; i < msgs.length; i++) {
      if (msgs[i].from !== 'customer') continue;
      const repliers = new Set();
      for (let j = i + 1; j < msgs.length && msgs[j].from !== 'customer'; j++) repliers.add(msgs[j].staff);
      if (repliers.size > 1) { duplicate = true; break; }
    }

    const last = msgs[msgs.length - 1];
    const awaiting = last.from === 'customer';
    // Oldest customer message that is still unanswered (drives SLA).
    let waitingSince = null;
    if (awaiting) {
      for (let i = msgs.length - 1; i >= 0 && msgs[i].from === 'customer'; i--) waitingSince = msgs[i].at;
    }

    const age = cfg.asOf - firstIn.at;
    return {
      id: thread.id,
      category: thread.category || null,
      firstIn: firstIn.at,
      firstReplyBy: firstReply ? firstReply.staff : null,
      frtMinutes: replyAt === null ? null : businessMinutes(firstIn.at, replyAt, cfg.hours),
      frtClockHours: replyAt === null ? null : (replyAt - firstIn.at) / HOUR,
      // Only judge a thread at 24h/48h once it is at least that old.
      unanswered24: age >= 24 * HOUR ? (replyAt === null || replyAt - firstIn.at > 24 * HOUR) : null,
      unanswered48: age >= 48 * HOUR ? (replyAt === null || replyAt - firstIn.at > 48 * HOUR) : null,
      neverAnswered: replyAt === null,
      duplicate,
      awaiting,
      waitingSince,
      staffReplies: msgs.filter(m => m.from === 'staff').map(m => m.staff),
    };
  }

  /** Inbox health check over a lookback window ending at cfg.asOf. */
  function audit(threads, cfg) {
    const from = cfg.asOf - (cfg.lookbackDays || 90) * DAY;
    const rows = threads.map(t => analyzeThread(t, cfg)).filter(r => r && r.firstIn >= from && r.firstIn <= cfg.asOf);
    const answered = rows.filter(r => r.frtMinutes !== null);
    const frt = answered.map(r => r.frtMinutes / 60);
    const judged24 = rows.filter(r => r.unanswered24 !== null);
    const judged48 = rows.filter(r => r.unanswered48 !== null);
    const pct = (num, den) => den.length ? num.length / den.length : null;

    const byCategory = {};
    rows.forEach(r => {
      const k = r.category || 'uncategorized';
      (byCategory[k] = byCategory[k] || []).push(r);
    });
    const categories = Object.keys(byCategory).map(k => {
      const rs = byCategory[k];
      const f = rs.filter(r => r.frtMinutes !== null).map(r => r.frtMinutes / 60);
      const j48 = rs.filter(r => r.unanswered48 !== null);
      return {
        category: k, threads: rs.length,
        medianFrtHours: f.length ? round(median(f), 2) : null,
        p90FrtHours: f.length ? round(quantile(f, 0.9), 2) : null,
        unanswered48: j48.length ? round(j48.filter(r => r.unanswered48).length / j48.length, 4) : null,
        duplicateRate: round(rs.filter(r => r.duplicate).length / rs.length, 4),
      };
    }).sort((a, b) => b.threads - a.threads);

    // Arrival heatmap: weekday (0=Sun) × hour of first customer message.
    const heat = Array.from({ length: 7 }, () => Array(24).fill(0));
    rows.forEach(r => { const d = new Date(r.firstIn); heat[d.getUTCDay()][d.getUTCHours()]++; });
    const outsideHours = rows.filter(r => {
      const d = new Date(r.firstIn), h = cfg.hours[d.getUTCDay()];
      return !h || d.getUTCHours() < h[0] || d.getUTCHours() >= h[1];
    }).length;

    // Who carries the inbox: share of all staff replies.
    const replyCounts = {};
    rows.forEach(r => r.staffReplies.forEach(s => { replyCounts[s] = (replyCounts[s] || 0) + 1; }));
    const totalReplies = Object.values(replyCounts).reduce((a, b) => a + b, 0);
    const staff = Object.keys(replyCounts).map(s => ({ staff: s, replies: replyCounts[s], share: round(replyCounts[s] / totalReplies, 4) }))
      .sort((a, b) => b.replies - a.replies);

    return {
      window: { from, to: cfg.asOf, days: cfg.lookbackDays || 90 },
      threads: rows.length,
      answered: answered.length,
      medianFrtHours: frt.length ? round(median(frt), 2) : null,
      p90FrtHours: frt.length ? round(quantile(frt, 0.9), 2) : null,
      unanswered24: pct(judged24.filter(r => r.unanswered24), judged24),
      unanswered48: pct(judged48.filter(r => r.unanswered48), judged48),
      neverAnswered: rows.filter(r => r.neverAnswered).length,
      awaitingNow: rows.filter(r => r.awaiting).length,
      duplicateRate: rows.length ? rows.filter(r => r.duplicate).length / rows.length : null,
      outsideHoursShare: rows.length ? outsideHours / rows.length : null,
      categories, heat, staff, rows,
    };
  }

  /** Plain-language findings, generated from the audit (never hard-coded). */
  function findings(a, labels, exclude) {
    const name = k => (labels && labels[k]) || k;
    const out = [];
    // `exclude` drops internal mail (e.g. vendors) from customer-facing comparisons.
    const cats = a.categories.filter(c => c.medianFrtHours !== null && c.threads >= 10 && !(exclude || []).includes(c.category));
    if (cats.length >= 2) {
      const slow = cats.reduce((x, y) => (y.medianFrtHours > x.medianFrtHours ? y : x));
      const fast = cats.reduce((x, y) => (y.medianFrtHours < x.medianFrtHours ? y : x));
      // The ratio is computed from the medians exactly as displayed (1 decimal),
      // so a reader can check it: 18.7 / 1.2 = 15.6, not the unrounded 15.2.
      const slowShown = round(slow.medianFrtHours, 1), fastShown = round(fast.medianFrtHours, 1);
      if (fastShown > 0 && slowShown / fastShown >= 1.5) {
        const cap = s => s.charAt(0).toUpperCase() + s.slice(1);
        out.push({ kind: 'slowest', ratio: round(slowShown / fastShown, 1), text: `${cap(name(slow.category))} wait ${round(slowShown / fastShown, 1)}× longer for a first reply than ${name(fast.category)} (median ${slowShown} vs ${fastShown} business hours).` });
      }
      const dropped = cats.filter(c => c.unanswered48 !== null).sort((x, y) => y.unanswered48 - x.unanswered48)[0];
      if (dropped && dropped.unanswered48 >= 0.1) {
        out.push({ kind: 'dropped', text: `${round(dropped.unanswered48 * 100)}% of ${name(dropped.category)} had no reply within 48 hours, the worst of any category.` });
      }
      const dup = cats.slice().sort((x, y) => y.duplicateRate - x.duplicateRate)[0];
      if (dup && dup.duplicateRate >= 0.05) {
        out.push({ kind: 'duplicate', text: `${round(dup.duplicateRate * 100)}% of ${name(dup.category)} got replies from two or more staff: nobody could see who already had it.` });
      }
    }
    if (a.staff.length && a.staff[0].share >= 0.4) {
      out.push({ kind: 'load', text: `${a.staff[0].staff} sent ${round(a.staff[0].share * 100)}% of all replies. The inbox depends on one person.` });
    }
    if (a.outsideHoursShare !== null && a.outsideHoursShare >= 0.25) {
      out.push({ kind: 'afterhours', text: `${round(a.outsideHoursShare * 100)}% of emails arrive outside staffed hours, so they start the next day already waiting.` });
    }
    return out;
  }

  /* ------------------------------------------------ ownership and SLAs */

  /**
   * Open-queue status at `now`. routing: {category: {owner, slaHours}} in business hours.
   * claims: {threadId: owner} overrides the routing default.
   */
  function queue(threads, cfg, claims, now) {
    now = now || cfg.asOf;
    const soonMin = (cfg.dueSoonHours || 1) * 60;
    return threads.map(t => analyzeThread(t, Object.assign({}, cfg, { asOf: now })))
      .filter(r => r && r.awaiting && r.waitingSince <= now)
      .map(r => {
        const route = (cfg.routing && cfg.routing[r.category]) || cfg.defaultRoute || { owner: null, slaHours: 8 };
        const owner = (claims && claims[r.id]) || null;
        const due = addBusinessMinutes(r.waitingSince, route.slaHours * 60, cfg.hours);
        const leftMin = due > now ? businessMinutes(now, due, cfg.hours) : -businessMinutes(due, now, cfg.hours);
        // Requests older than staleDays are a one-time backlog to clean up, not
        // something to nudge about every day.
        const stale = cfg.staleDays && now - r.waitingSince > cfg.staleDays * DAY;
        const status = stale ? 'stale' : leftMin < 0 ? 'overdue' : leftMin <= soonMin ? 'due-soon' : 'on-track';
        const unclaimedHours = owner ? 0 : (now - r.waitingSince) / HOUR;
        return {
          id: r.id, category: r.category, owner, suggestedOwner: route.owner, slaHours: route.slaHours,
          waitingSince: r.waitingSince, due, businessMinutesLeft: round(leftMin), status, unclaimedHours: round(unclaimedHours, 1),
        };
      })
      .sort((a, b) => (a.status === 'stale') - (b.status === 'stale') || a.due - b.due);
  }

  /**
   * Nudge plan: who gets an email right now. Owners get one digest of their
   * overdue threads; unclaimed threads older than cfg.escalateUnclaimedHours go
   * to the whole team. Batched per person to respect daily send limits.
   */
  function nudges(q, cfg) {
    const byPerson = {};
    const add = (who, item, why) => { (byPerson[who] = byPerson[who] || []).push({ id: item.id, why }); };
    q.forEach(item => {
      if (item.status === 'stale') return;
      if (item.owner && item.status === 'overdue') add(item.owner, item, 'overdue');
      if (!item.owner && item.unclaimedHours >= (cfg.escalateUnclaimedHours || 24)) cfg.team.forEach(p => add(p, item, 'unclaimed'));
    });
    return Object.keys(byPerson).map(p => ({ to: p, items: byPerson[p] }));
  }

  /** Weekly scorecard rows for the last n weeks (Monday-start weeks). */
  function weekly(threads, cfg, weeks) {
    const rows = [];
    const d = new Date(cfg.asOf);
    const monday = Date.UTC(d.getUTCFullYear(), d.getUTCMonth(), d.getUTCDate()) - ((d.getUTCDay() + 6) % 7) * DAY;
    for (let w = (weeks || 6) - 1; w >= 0; w--) {
      const start = monday - w * 7 * DAY, end = Math.min(start + 7 * DAY, cfg.asOf);
      const a = audit(threads, Object.assign({}, cfg, { asOf: end, lookbackDays: (end - start) / DAY }));
      rows.push({ weekOf: start, threads: a.threads, medianFrtHours: a.medianFrtHours, unanswered48: a.unanswered48, duplicateRate: a.duplicateRate });
    }
    return rows;
  }

  /* ----------------------------------------------------------- triage */

  /**
   * Rules-first triage. rules: {category: [[regexSource, weight], ...]}.
   * Confidence = top score / (top + runner-up). Below `threshold` goes to a
   * human instead of being auto-routed.
   */
  function compileRules(rules) {
    const out = {};
    Object.keys(rules).forEach(k => { out[k] = rules[k].map(([src, w]) => [new RegExp(src, 'i'), w, src]); });
    return out;
  }

  function classify(text, compiled, threshold) {
    const scores = [];
    Object.keys(compiled).forEach(k => {
      let s = 0; const hits = [];
      compiled[k].forEach(([re, w, src]) => { if (re.test(text)) { s += w; hits.push(src); } });
      scores.push({ category: k, score: s, hits });
    });
    scores.sort((a, b) => b.score - a.score);
    const top = scores[0], second = scores[1];
    if (!top || top.score <= 0) return { category: null, confidence: 0, needsHuman: true, scores };
    const confidence = top.score / (top.score + Math.max(0, second ? second.score : 0));
    return { category: top.category, confidence: round(confidence, 3), needsHuman: confidence < (threshold || 0.6), hits: top.hits, scores };
  }

  /** Accuracy report against hand labels. items: [{text, label}] */
  function evaluate(items, compiled, threshold, categories) {
    const cats = categories || Object.keys(compiled);
    const matrix = {};
    cats.forEach(a => { matrix[a] = {}; cats.concat(['(human)']).forEach(b => { matrix[a][b] = 0; }); });
    let correct = 0, auto = 0, autoCorrect = 0;
    const errors = [];
    items.forEach(it => {
      const r = classify(it.text, compiled, threshold);
      const predicted = r.needsHuman ? '(human)' : r.category;
      matrix[it.label][predicted] = (matrix[it.label][predicted] || 0) + 1;
      if (!r.needsHuman) {
        auto++;
        if (r.category === it.label) autoCorrect++; else errors.push({ text: it.text, label: it.label, predicted: r.category, confidence: r.confidence });
      }
      if (r.category === it.label) correct++;
    });
    const perClass = cats.map(c => {
      const tp = matrix[c][c];
      const predictedC = cats.reduce((s, a) => s + (matrix[a][c] || 0), 0);
      const actualC = cats.concat(['(human)']).reduce((s, b) => s + (matrix[c][b] || 0), 0);
      return { category: c, support: actualC, precision: predictedC ? round(tp / predictedC, 3) : null, recall: actualC ? round(tp / actualC, 3) : null };
    });
    return {
      n: items.length,
      topLabelAccuracy: round(correct / items.length, 3),       // if every email were auto-routed
      autoRouted: auto, autoRate: round(auto / items.length, 3),
      autoPrecision: auto ? round(autoCorrect / auto, 3) : null, // accuracy of what actually gets auto-routed
      humanQueue: items.length - auto,
      perClass, matrix, errors,
    };
  }

  const api = { HOUR, MIN, DAY, median, quantile, businessMinutes, addBusinessMinutes, analyzeThread, audit, findings, queue, nudges, weekly, compileRules, classify, evaluate };
  if (typeof module !== 'undefined' && module.exports) module.exports = api;
  else root.InboxCore = api;
})(typeof globalThis !== 'undefined' ? globalThis : this);

// Converts a real Date to studio wall-clock ms (the core's time convention).
function toWallClock(date, tz) {
  const s = Utilities.formatDate(date, tz, "yyyy-MM-dd'T'HH:mm:ss");
  return Date.UTC(+s.slice(0, 4), +s.slice(5, 7) - 1, +s.slice(8, 10), +s.slice(11, 13), +s.slice(14, 16), +s.slice(17, 19));
}

function addressOf(from) {
  const m = /<([^>]+)>/.exec(from);
  return (m ? m[1] : from).trim().toLowerCase();
}

/**
 * Turns one GmailThread into {id, category, subject?, messages:[{from, staff, at}]}.
 * Staff = any team member's address or the shared address itself.
 */
function convertThread(thread, st, rules) {
  const byEmail = {};
  st.team.forEach(t => { byEmail[t.email] = t.name; });
  const messages = thread.getMessages().map(m => {
    const addr = addressOf(m.getFrom());
    const isStaff = addr === st.shared || Object.prototype.hasOwnProperty.call(byEmail, addr);
    return { from: isStaff ? 'staff' : 'customer', staff: isStaff ? (byEmail[addr] || 'Shared inbox') : null, at: toWallClock(m.getDate(), st.tz) };
  });
  const firstIn = thread.getMessages().find(m => {
    const a = addressOf(m.getFrom());
    return a !== st.shared && !Object.prototype.hasOwnProperty.call(byEmail, a);
  });
  let category = null;
  if (firstIn && rules) {
    // Body is used in memory only; it is never written to the Sheet.
    const text = thread.getFirstMessageSubject() + '\n' + String(firstIn.getPlainBody()).slice(0, 600);
    const r = InboxCore.classify(text, rules, st.triageThreshold);
    category = r.needsHuman ? 'needs-review' : r.category;
  }
  const out = { id: thread.getId(), category, messages: messages.sort((a, b) => a.at - b.at) };
  if (st.storeSubjects) out.subject = thread.getFirstMessageSubject();
  return out;
}

function fetchThreads(st) {
  const rules = InboxCore.compileRules(TRIAGE_RULES);
  const query = `to:${st.shared} newer_than:${st.lookbackDays}d -in:spam -in:trash`;
  const threads = [];
  for (let start = 0; start < 1500; start += 100) {
    const page = GmailApp.search(query, start, 100);
    page.forEach(t => threads.push(convertThread(t, st, rules)));
    if (page.length < 100) break;
  }
  return threads;
}

/**
 * Generates 90+ days of synthetic shared-inbox email for "Cedar & Iron
 * Fitness" (fictional), deterministically from a seed.
 *
 * Behaviour baked into the model (what a real audit would have to discover):
 *  - many emails arrive before/after staffed hours (members email around workouts)
 *  - uncomfortable requests (cancellations, freezes, vendors) wait longest and
 *    are most often never answered
 *  - easy requests (trials, schedule) sometimes get two replies because nobody
 *    can see who already answered
 *  - one person (the owner) carries most money-related email
 * Usage: node data/generate.js  → writes web/data/inbox.json
 */
const fs = require('fs');
const path = require('path');
const C = require('../core/inbox-core.js');
const { CATEGORIES, PHRASINGS, FOLLOW_UPS, STAFF_REPLIES } = require('./phrasings.js');

const AS_OF = Date.UTC(2026, 9, 22, 16, 0); // Thu Oct 22, 2026, 4:00pm studio time
const DAYS = 95;
const HOURS = { 0: null, 1: [9, 18], 2: [9, 18], 3: [9, 18], 4: [9, 18], 5: [9, 18], 6: [9, 13] };
const STAFF = ['Maya', 'Jordan', 'Priya', 'Sam'];

function mulberry32(a) {
  return function () {
    a |= 0; a = (a + 0x6D2B79F5) | 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

function generate(seed) {
  const rnd = mulberry32(seed || 20261022);
  const pick = a => a[Math.floor(rnd() * a.length)];
  const weighted = obj => {
    const keys = Object.keys(obj), total = keys.reduce((s, k) => s + obj[k], 0);
    let x = rnd() * total;
    for (const k of keys) { x -= obj[k]; if (x <= 0) return k; }
    return keys[keys.length - 1];
  };
  const normal = () => Math.sqrt(-2 * Math.log(1 - rnd())) * Math.cos(2 * Math.PI * rnd());
  const lognormal = (medianMin, sigma) => Math.max(3, medianMin * Math.exp(sigma * normal()));

  const MIX = { trial: 22, schedule: 18, billing: 14, cancel: 10, freeze: 9, training: 10, vendor: 7, general: 10 };
  // Arrival hour weights: peaks before work and after evening workouts.
  const HOUR_W = [0.1, 0.05, 0.05, 0.05, 0.1, 0.3, 0.8, 1.2, 1.6, 1.8, 1.7, 1.6, 1.8, 1.7, 1.5, 1.4, 1.4, 1.3, 1.2, 1.2, 1.3, 1.1, 0.6, 0.3];
  // Who answers what (before any ownership rules exist).
  const RESPONDER = {
    trial: { Jordan: 5, Priya: 3, Maya: 2 }, schedule: { Jordan: 5, Priya: 4, Maya: 1 },
    billing: { Maya: 8, Jordan: 2 }, cancel: { Maya: 9, Jordan: 1 }, freeze: { Maya: 7, Jordan: 3 },
    training: { Sam: 6, Maya: 3, Jordan: 1 }, vendor: { Maya: 10 }, general: { Jordan: 4, Priya: 4, Maya: 2 },
  };
  // Median first-reply delay in BUSINESS minutes, and chance of never replying.
  const DELAY = { trial: 70, schedule: 110, billing: 280, cancel: 900, freeze: 520, training: 300, vendor: 1300, general: 380 };
  const NEVER = { trial: 0.02, schedule: 0.04, billing: 0.03, cancel: 0.16, freeze: 0.1, training: 0.04, vendor: 0.22, general: 0.1 };
  const DUP = { trial: 0.15, schedule: 0.09, general: 0.05 };
  const GREET = ['', '', 'Hi, ', 'Hello, ', 'Hey there, ', 'Good morning, '];
  const SIGN = ['', '', ' Thanks!', ' Thank you.', ' Thx', ' Appreciate it.'];
  const NOISE = ['', '', '', '', " I've been a member for two years.", ' Love the classes by the way.', ' Sorry if this is the wrong email.', ' I emailed last week too.'];
  const FIRST = ['Alex', 'Sam', 'Jamie', 'Taylor', 'Morgan', 'Riley', 'Casey', 'Avery', 'Quinn', 'Rowan', 'Dakota', 'Reese', 'Emerson', 'Skyler', 'Kai', 'Noor', 'Lena', 'Mateo', 'Hana', 'Theo'];
  const LAST = ['Nguyen', 'Garcia', 'Okafor', 'Larsen', 'Haddad', 'Ito', 'Moreno', 'Kowalski', 'Silva', 'Fischer', 'Achebe', 'Lindqvist', 'Patel', 'Duval', 'Tanaka', 'Romero'];

  const threads = [];
  let n = 0;
  for (let day = DAYS; day >= 0; day--) {
    const dayStart = Date.UTC(2026, 9, 22) - day * C.DAY;
    const dow = new Date(dayStart).getUTCDay();
    const mean = dow === 0 ? 4 : dow === 6 ? 5 : dow === 1 ? 7.5 : 6;
    // Poisson draw
    let k = 0, p = Math.exp(-mean), s = p; const u = rnd();
    while (u > s) { k++; p *= mean / k; s += p; }
    for (let i = 0; i < k; i++) {
      const hour = Number(weighted(Object.fromEntries(HOUR_W.map((w, h) => [h, w]))));
      const at = dayStart + hour * C.HOUR + Math.floor(rnd() * 60) * C.MIN;
      if (at > AS_OF) continue;
      const category = weighted(MIX);
      const split = rnd() < 0.5 ? 'dev' : 'test';
      const [subject, body] = pick(PHRASINGS[category][split]);
      const first = pick(FIRST), last = pick(LAST);
      let text = pick(GREET) + body + pick(NOISE) + pick(SIGN);
      if (rnd() < 0.12) text = text.toLowerCase();
      const thread = {
        id: 'T' + String(++n).padStart(4, '0'), category, split, subject,
        customer: { name: first + ' ' + last, email: `${first}.${last}@mail.example`.toLowerCase() },
        messages: [{ from: 'customer', at, text }],
      };

      // First reply
      const responder = weighted(RESPONDER[category]);
      let replyAt = rnd() < NEVER[category] ? null : C.addBusinessMinutes(at, lognormal(DELAY[category], 0.9), HOURS);
      replyAt = replyAt === null ? null : Math.round(replyAt / C.MIN) * C.MIN;

      // Customer chases if nothing came back within a day
      if ((replyAt === null || replyAt - at > 24 * C.HOUR) && rnd() < 0.45) {
        const fu = at + (26 + Math.floor(rnd() * 14)) * C.HOUR;
        if (fu <= AS_OF && (replyAt === null || fu < replyAt)) thread.messages.push({ from: 'customer', at: fu, text: pick(FOLLOW_UPS) });
      }
      if (replyAt !== null && replyAt <= AS_OF) {
        thread.messages.push({ from: 'staff', staff: responder, at: replyAt, text: STAFF_REPLIES[category] });
        // Second person answers too, unaware someone already did
        if (DUP[category] && rnd() < DUP[category]) {
          const other = pick(STAFF.filter(x => x !== responder && x !== 'Sam'));
          const dupAt = replyAt + (5 + Math.floor(rnd() * 40)) * C.MIN;
          if (dupAt <= AS_OF) thread.messages.push({ from: 'staff', staff: other, at: dupAt, text: STAFF_REPLIES[category] });
        }
        // Conversation continues
        if (rnd() < 0.3) {
          const back = replyAt + (1 + Math.floor(rnd() * 30)) * C.HOUR;
          if (back <= AS_OF) {
            thread.messages.push({ from: 'customer', at: back, text: 'Thanks. One more question about this.' });
            if (rnd() < 0.85) {
              const again = Math.round(C.addBusinessMinutes(back, lognormal(DELAY[category] / 2, 0.8), HOURS) / C.MIN) * C.MIN;
              if (again <= AS_OF) thread.messages.push({ from: 'staff', staff: responder, at: again, text: 'Answered below.' });
            }
          }
        }
      }
      thread.messages.sort((a, b) => a.at - b.at);
      threads.push(thread);
    }
  }
  return threads;
}
