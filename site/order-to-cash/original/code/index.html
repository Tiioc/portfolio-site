  • Data.gs: Seeded generator for the sample orders, with the planted errors and an answer key, plus the same-name test case.

  • appsscript.json: Permissions: this spreadsheet plus sending mail.

// Converts a 1-based column index to its A1 letter.
function colLetter(n) {
  let s = '';
  while (n > 0) {
    const m = (n - 1) % 26;
    s = String.fromCharCode(65 + m) + s;
    n = Math.floor((n - 1) / 26);
  }
  return s;
}

// Builds {key: letter} and {key: index} lookups from a column spec.
function indexColumns(spec) {
  const L = {}, I = {};
  spec.forEach((c, i) => { L[c.key] = colLetter(i + 1); I[c.key] = i + 1; });
  return { L, I };
}

// Returns the formula for an Orders cell at row r. `o` = Orders column letters.
function orderFormula(key, r) {
  const o = OC.L;
  const c = k => o[k] + r;
  const pay = k => `Payments!$${PC.L[k]}:$${PC.L[k]}`;
  switch (key) {
    case 'customer':
      return `=IF(${c('cust_id')}="","",IFERROR(VLOOKUP(${c('cust_id')},Customers!$A:$B,2,FALSE),"Unknown customer"))`;
    case 'csm':
      return `=IF(${c('cust_id')}="","",IFERROR(VLOOKUP(${c('cust_id')},Customers!$A:$E,5,FALSE),""))`;
    case 'product':
      return `=IF(${c('sku')}="","",IFERROR(VLOOKUP(${c('sku')},Products!$A:$B,2,FALSE),"Unknown SKU"))`;
    case 'order_value':
      return `=IF(${c('qty')}="","",${c('qty')}*${c('unit_price')})`;
    case 'contract_price':
      return `=IFERROR(INDEX('Contract Prices'!$G:$G,MATCH(${c('cust_id')}&"|"&${c('sku')},'Contract Prices'!$A:$A,0)),"")`;
    case 'contract_terms':
      return `=IF(${c('cust_id')}="","",IFERROR(VLOOKUP(${c('cust_id')},Customers!$A:$G,7,FALSE),""))`;
    case 'val_issues':
      return `=IF(${c('order_id')}="","",TEXTJOIN("; ",TRUE,` +
        `IF(${c('product')}="Unknown SKU","Unknown SKU",IF(${c('contract_price')}="","Product not on contract","")),` +
        `IF(AND(N(${c('contract_price')})>0,ABS(${c('unit_price')}-N(${c('contract_price')}))>0.005),"Price mismatch ("&TEXT(${c('unit_price')}-N(${c('contract_price')}),"+$#,##0;-$#,##0")&"/unit vs contract)",""),` +
        `IF(OR(${c('bill_name')}="",${c('bill_email')}="",ISERROR(SEARCH("@",${c('bill_email')}))),"Missing/invalid billing contact",""),` +
        `IF(${c('terms')}<>${c('contract_terms')},"Terms "&${c('terms')}&" ≠ contract "&${c('contract_terms')},"")))`;
    case 'val_status':
      return `=IF(${c('order_id')}="","",IF(${c('val_issues')}="","${PASS}","${HOLD}"))`;
    case 'due_date':
      return `=IF(${c('invoiced')}="","",${c('invoiced')}+IFERROR(VALUE(SUBSTITUTE(${c('terms')},"Net ","")),30))`;
    case 'stage':
      return `=IF(${c('order_id')}="","",IF(${c('paid')}<>"","Paid",IF(${c('invoiced')}<>"","Invoiced",IF(${c('activated')}<>"","Activated",IF(${c('shipped')}<>"","Shipped",IF(${c('acct_setup')}<>"","Account Set Up",IF(${c('validated')}<>"","Validated","Received")))))))`;
    case 'stage_date':
      return `=IF(${c('order_id')}="","",MAX(${c('received')},${o.validated}${r}:${o.paid}${r}))`;
    case 'days_in_stage':
      return `=IF(OR(${c('order_id')}="",${c('stage')}="Paid"),"",AS_OF-${c('stage_date')})`;
    case 'sla':
      return `=IFERROR(VLOOKUP(${c('stage')},SLA_TABLE,2,FALSE),"")`;
    case 'sla_breach':
      return `=IF(OR(${c('sla')}="",${c('days_in_stage')}=""),"",IF(${c('days_in_stage')}>${c('sla')},"${BREACH}",""))`;
    case 'activation_watch':
      return `=IF(AND(${c('stage')}="Shipped",AS_OF-N(${c('shipped')})>STALL_DAYS),"Stalled "&(AS_OF-${c('shipped')})&"d since ship — notify CSM "&${c('csm')},"")`;
    case 'cycle_days':
      return `=IF(${c('activated')}="","",${c('activated')}-${c('received')})`;
    case 'amt_paid':
      return `=IF(${c('invoice_no')}="",0,SUMIFS(${pay('applied')},${pay('matched_inv')},${c('invoice_no')}))`;
    case 'balance':
      return `=IF(${c('invoice_no')}="","",ROUND(${c('order_value')}-${c('amt_paid')},2))`;
    case 'dpd':
      return `=IF(OR(${c('due_date')}="",N(${c('balance')})<=0.005),"",MAX(0,AS_OF-${c('due_date')}))`;
    case 'aging':
      return `=IF(OR(${c('due_date')}="",N(${c('balance')})<=0.005),"",IF(${c('dpd')}=0,"Current",IF(${c('dpd')}<=30,"1–30",IF(${c('dpd')}<=60,"31–60",IF(${c('dpd')}<=90,"61–90","90+")))))`;
    case 'risk_reasons':
      return `=IF(${c('order_id')}="","",TEXTJOIN("; ",TRUE,` +
        `IF(${c('val_status')}="${HOLD}","Intake hold",""),` +
        `IF(${c('activation_watch')}<>"","Stalled activation",IF(${c('sla_breach')}="${BREACH}","SLA breach: "&${c('stage')},"")),` +
        `IF(N(${c('dpd')})>0,"Past due "&${c('dpd')}&"d","")))`;
    case 'rar':
      return `=IF(OR(${c('order_id')}="",${c('risk_reasons')}=""),0,IF(${c('invoice_no')}="",${c('order_value')},N(${c('balance')})))`;
    case 'match_key':
      return `=IF(N(${c('balance')})>0.005,${c('cust_id')}&"|"&TEXT(${c('balance')},"0.00"),"")`;
  }
  throw new Error('No formula for Orders.' + key);
}

// Returns the formula for a Payments cell at row r.
function paymentFormula(key, r) {
  const p = PC.L;
  const c = k => p[k] + r;
  const ord = k => `Orders!$${OC.L[k]}:$${OC.L[k]}`;
  const above = k => `${p[k]}$1:${p[k]}${r - 1}`;
  switch (key) {
    case 'customer':
      return `=IF(${c('cust_id')}="","",IFERROR(VLOOKUP(${c('cust_id')},Customers!$A:$B,2,FALSE),"Unknown customer"))`;
    case 'matched_inv':
      return `=IF(${c('remit_ref')}="","",IFERROR(IF(INDEX(${ord('cust_id')},MATCH(${c('remit_ref')},${ord('invoice_no')},0))=${c('cust_id')},${c('remit_ref')},""),""))`;
    case 'inv_amount':
      return `=IF(${c('matched_inv')}="","",INDEX(${ord('order_value')},MATCH(${c('matched_inv')},${ord('invoice_no')},0)))`;
    case 'prior_applied':
      return `=IF(${c('matched_inv')}="","",SUMIFS(${above('applied')},${above('matched_inv')},${c('matched_inv')}))`;
    case 'applied':
      return `=IF(${c('matched_inv')}="",0,ROUND(MAX(0,MIN(${c('amount')},${c('inv_amount')}-${c('prior_applied')})),2))`;
    case 'unapplied':
      return `=ROUND(${c('amount')}-${c('applied')},2)`;
    case 'match_status':
      return `=IF(${c('pay_id')}="","",IF(${c('matched_inv')}<>"",` +
        `IF(${c('unapplied')}<=0.005,IF(${c('applied')}<${c('inv_amount')}-${c('prior_applied')}-0.005,"Short-paid (partial)","Applied"),` +
        `IF(${c('applied')}=0,"Duplicate payment","Overpayment")),` +
        `IF(${c('remit_ref')}="","No remittance ref",IF(COUNTIF(${ord('invoice_no')},${c('remit_ref')})>0,"Customer mismatch","No matching invoice"))))`;
    case 'recon_action':
      return `=IF(${c('unapplied')}<=0.005,"",IF(${c('match_status')}="Duplicate payment","Invoice already paid — confirm duplicate with customer; refund or hold as credit",` +
        `IF(${c('match_status')}="Overpayment","Paid more than invoiced — confirm with customer; refund or hold as credit",` +
        `IFERROR("Likely match: "&INDEX(${ord('invoice_no')},MATCH(${c('cust_id')}&"|"&TEXT(${c('amount')},"0.00"),${ord('match_key')},0))&" (same customer, same open amount) — verify and re-apply",` +
        `"Research — no open invoice for this customer and amount; contact customer AP"))))`;
  }
  throw new Error('No formula for Payments.' + key);
}

function mulberry32(a) {
  return function () {
    a |= 0; a = (a + 0x6D2B79F5) | 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

function generateDemoData(asOf, seed) {
  asOf = asOf || DEMO_AS_OF;
  const rnd = mulberry32(seed || DEMO_SEED);
  const ri = (a, b) => a + Math.floor(rnd() * (b - a + 1)); // inclusive int
  const pick = arr => arr[Math.floor(rnd() * arr.length)];
  const listPrice = {};
  PRODUCTS.forEach(p => { listPrice[p[0]] = p[2]; });
  const productName = {};
  PRODUCTS.forEach(p => { productName[p[0]] = p[1]; });

  /* ---------- customers + contract prices */
  const customers = CUSTOMER_SEED.map((c, i) => {
    const slug = c[0].toLowerCase().replace(/[^a-z]/g, '').slice(0, 16);
    const first = pick(FIRST), last = pick(LAST);
    const csm = CSMS[i % CSMS.length];
    const contractSkus = {};
    SEGMENT_SKUS[c[1]].forEach(sku => {
      const disc = sku === 'SVC-ONB' ? ri(0, 2) * 5 : ri(3, 18); // percent
      contractSkus[sku] = Math.round(listPrice[sku] * (1 - disc / 100));
    });
    return {
      id: 'C-' + (101 + i), name: c[0], segment: c[1], region: c[2],
      csm: csm[0], csmEmail: csm[1],
      terms: pick(['Net 30', 'Net 30', 'Net 30', 'Net 45', 'Net 60']),
      billName: first + ' ' + last,
      billEmail: 'ap.' + first.toLowerCase() + '@' + slug + '.example',
      skus: contractSkus,
    };
  });
  const prices = [];
  customers.forEach(c => Object.keys(c.skus).forEach(sku => {
    prices.push([c.id, c.name, sku, productName[sku], listPrice[sku], c.skus[sku]]);
  }));

  const qtyFor = sku => sku.startsWith('ACC') ? ri(10, 120)
    : sku === 'SVC-ONB' ? ri(1, 8)
    : sku === 'TRM-AVN' ? ri(1, 6)
    : sku === 'TRM-MAR' ? ri(2, 14)
    : ri(5, 60);
  const termDays = t => parseInt(String(t).replace('Net ', ''), 10) || 30;

  /* ---------- scenario plan */
  const N = 150;
  const scen = [];
  const push = (s, n) => { for (let i = 0; i < n; i++) scen.push(s); };
  push('hold', 22);
  push('stalled', 7);
  push('sla_validated', 3);   // account set-up backlog
  push('sla_setup', 2);       // set up but not shipped
  push('sla_activated', 2);   // activated but never invoiced = billing miss
  push('normal', N - scen.length);
  // shuffle
  for (let i = scen.length - 1; i > 0; i--) { const j = Math.floor(rnd() * (i + 1)); [scen[i], scen[j]] = [scen[j], scen[i]]; }

  const holdIssues = [];
  ['price', 'price', 'price', 'price', 'price', 'price', 'price', 'price',
    'not_on_contract', 'not_on_contract', 'not_on_contract', 'unknown_sku', 'unknown_sku',
    'billing', 'billing', 'billing', 'billing', 'billing',
    'terms', 'terms', 'terms', 'terms'].forEach(x => holdIssues.push([x]));
  holdIssues[1].push('billing');   // two orders carry two problems
  holdIssues[14].push('terms');

  const DUR = { val: [0, 2], setup: [1, 3], ship: [2, 6], act: [3, 12], inv: [1, 3] };
  const d = k => ri(DUR[k][0], DUR[k][1]);

  /* ---------- orders */
  let holdIdx = 0;
  const orders = scen.map(s => {
    const c = pick(customers);
    const skus = Object.keys(c.skus);
    const sku = pick(skus);
    const o = {
      scenario: s, cust: c, cust_id: c.id, sku, qty: qtyFor(sku),
      unit_price: c.skus[sku], bill_name: c.billName, bill_email: c.billEmail, terms: c.terms,
      issues: [], received: null, validated: '', acct_setup: '', shipped: '', activated: '', invoiced: '', paid: '',
      invoice_no: '', actions: '', payBehavior: null,
    };
    if (s === 'hold') {
      o.received = asOf - ri(0, 20);
      o.issues = holdIssues[holdIdx++];
      o.issues.forEach(issue => applyIssue(o, issue));
    } else if (s === 'stalled') {
      o.shipped = asOf - ri(17, 52);
      backfill(o, 'shipped');
    } else if (s === 'sla_validated') {
      o.validated = asOf - ri(6, 11);
      backfill(o, 'validated');
    } else if (s === 'sla_setup') {
      o.acct_setup = asOf - ri(12, 21);
      backfill(o, 'acct_setup');
    } else if (s === 'sla_activated') {
      o.activated = asOf - ri(9, 16);
      backfill(o, 'activated');
    } else {
      o.received = asOf - ri(0, 210);
      forward(o);
    }
    return o;
  });

  function applyIssue(o, issue) {
    const c = o.cust;
    if (issue === 'price') {
      // Most common real-world miss: rep quotes list price instead of contract price.
      o.unit_price = rnd() < 0.6 ? listPrice[o.sku] : Math.round(c.skus[o.sku] * (1 + pick([-0.12, -0.08, 0.07, 0.1, 0.15])));
      if (o.unit_price === c.skus[o.sku]) o.unit_price += 50;
    } else if (issue === 'not_on_contract') {
      const off = PRODUCTS.map(p => p[0]).filter(s => !(s in c.skus));
      o.sku = pick(off); o.qty = qtyFor(o.sku); o.unit_price = listPrice[o.sku];
    } else if (issue === 'unknown_sku') {
      o.sku = pick(['TRM-HP2', 'TRM-STD-X', 'SVC-ONBD']); o.unit_price = c.skus[Object.keys(c.skus)[0]];
    } else if (issue === 'billing') {
      const k = ri(0, 2);
      if (k === 0) o.bill_name = '';
      else if (k === 1) o.bill_email = '';
      else o.bill_email = o.bill_email.replace('@', '.'); // malformed address
    } else if (issue === 'terms') {
      o.terms = pick(['Net 60', 'Net 90'].filter(t => t !== c.terms));
    }
  }

  // Walks back from a known stage date to fill earlier stages.
  function backfill(o, stageKey) {
    const order = ['received', 'validated', 'acct_setup', 'shipped', 'activated'];
    const back = { validated: 'val', acct_setup: 'setup', shipped: 'ship', activated: 'act' };
    for (let i = order.indexOf(stageKey); i > 0; i--) {
      o[order[i - 1]] = o[order[i]] - d(back[order[i]]);
    }
  }

  // Moves a clean order forward through the pipeline until it reaches as-of.
  function forward(o) {
    const steps = [['validated', 'val'], ['acct_setup', 'setup'], ['shipped', 'ship'], ['activated', 'act'], ['invoiced', 'inv']];
    let t = o.received;
    for (const [k, dk] of steps) {
      t += d(dk);
      if (t > asOf) return;
      o[k] = t;
    }
  }

  orders.sort((a, b) => a.received - b.received || (a.cust_id < b.cust_id ? -1 : 1));
  orders.forEach((o, i) => { o.order_id = 'SO-' + (10001 + i); });

  /* ---------- invoices */
  const invoiced = orders.filter(o => o.invoiced !== '').sort((a, b) => a.invoiced - b.invoiced);
  invoiced.forEach((o, i) => {
    o.invoice_no = 'INV-' + (20301 + i);
    o.value = o.qty * o.unit_price;
    o.due = o.invoiced + termDays(o.terms);
  });

  /* ---------- payment behaviour */
  const payments = [];
  const pay = (o, date, amount, ref, method) => payments.push({ cust_id: o ? o.cust_id : null, date, amount: Math.round(amount * 100) / 100, ref, method: method || pick(['ACH', 'ACH', 'Wire', 'Check']) });

  // Planted cash-application problems, chosen from invoices whose story makes sense.
  const paidCandidates = invoiced.filter(o => o.due - 12 <= asOf - 10);
  const pastDueCandidates = invoiced.filter(o => o.due < asOf - 5);
  const taken = new Set();
  const choose = (pool, n) => {
    const out = [];
    const avail = pool.filter(o => !taken.has(o));
    while (out.length < n && avail.length) {
      const o = avail.splice(Math.floor(rnd() * avail.length), 1)[0];
      taken.add(o); out.push(o);
    }
    return out;
  };
  choose(paidCandidates, 4).forEach(o => { o.payBehavior = 'duplicate'; });
  choose(paidCandidates, 2).forEach(o => { o.payBehavior = 'overpay'; });
  choose(pastDueCandidates, 3).forEach(o => { o.payBehavior = 'misref'; });
  choose(pastDueCandidates, 4).forEach(o => { o.payBehavior = 'partial'; });
  // Deliberately non-paying accounts, spread across aging buckets.
  const byAge = pastDueCandidates.filter(o => !taken.has(o)).sort((a, b) => a.due - b.due);
  const nonPay = [];
  [0.02, 0.1, 0.22, 0.4, 0.6, 0.78, 0.9].forEach(q => {
    const o = byAge[Math.floor(q * (byAge.length - 1))];
    if (o && !taken.has(o)) { taken.add(o); nonPay.push(o); o.payBehavior = 'nonpay'; }
  });

  invoiced.forEach(o => {
    if (!o.payBehavior) o.payBehavior = rnd() < 0.82 ? 'ontime' : 'late';
    let payDate;
    switch (o.payBehavior) {
      case 'ontime': payDate = o.due - ri(0, 12); break;
      case 'late': payDate = o.due + ri(4, 35); break;
      case 'duplicate': case 'overpay': payDate = Math.min(o.due - ri(0, 10), asOf - 10); break;
      case 'misref': payDate = Math.min(o.due + ri(0, 10), asOf - 2); break;
      case 'partial': payDate = Math.min(o.due + ri(-5, 5), asOf - 3); break;
      default: payDate = null;
    }
    if (payDate === null) return;
    payDate = Math.max(payDate, o.invoiced + 1);
    if (payDate > asOf) return; // not paid yet
    if (o.payBehavior === 'partial') {
      pay(o, payDate, Math.round(o.value * pick([0.5, 0.6, 0.75, 0.8])), o.invoice_no);
      return;
    }
    if (o.payBehavior === 'misref') {
      // Customer paid the right amount but referenced their own PO / mistyped the invoice.
      const bad = pick(['po', 'typo', 'blank']);
      const ref = bad === 'po' ? 'PO-' + ri(40000, 49999) : bad === 'typo' ? o.invoice_no.replace('INV-20', 'INV-2O') : '';
      pay(o, payDate, o.value, ref);
      return;
    }
    if (o.payBehavior === 'overpay') {
      // Paid list price instead of contract price (or a rounding overage when they're equal).
      const listTotal = o.qty * (listPrice[o.sku] || o.unit_price);
      pay(o, payDate, listTotal > o.value ? listTotal : o.value + 250, o.invoice_no);
      o.paid = payDate;
      return;
    }
    pay(o, payDate, o.value, o.invoice_no);
    o.paid = payDate;
    if (o.payBehavior === 'duplicate') pay(o, Math.min(payDate + ri(4, 9), asOf), o.value, o.invoice_no);
  });

  // Two receipts nobody can identify (no invoice for that customer at that amount).
  const mystery = [pick(customers), pick(customers)];
  pay({ cust_id: mystery[0].id }, asOf - ri(3, 20), 18450, '', 'Wire');
  pay({ cust_id: mystery[1].id }, asOf - ri(3, 20), 7342.5, 'DEPOSIT', 'Check');

  payments.sort((a, b) => a.date - b.date);
  payments.forEach((p, i) => { p.pay_id = 'PMT-' + (50001 + i); });

  /* ---------- actions-taken notes (some intentionally blank) */
  const fmt = s => serialToIso(s);
  orders.forEach(o => {
    const notes = [];
    if (o.scenario === 'stalled' && rnd() < 0.75) {
      notes.push(fmt(o.shipped + 15) + ' — Emailed site contact to confirm install date');
      if (asOf - o.shipped > 25) notes.push(fmt(o.shipped + 22) + ' — Called site lead, left voicemail');
    }
    if (o.scenario === 'hold' && rnd() < 0.6) notes.push(fmt(o.received) + ' — Returned order to account rep for correction');
    if (o.scenario === 'sla_validated') notes.push(fmt(o.validated + 3) + ' — Account set-up ticket opened with Billing Ops');
    if (o.scenario === 'sla_setup') notes.push(fmt(o.acct_setup + 5) + ' — Warehouse reports backorder on terminal kits');
    if (o.payBehavior === 'nonpay' && o.due < asOf - 20 && rnd() < 0.8) {
      notes.push(fmt(o.due + 7) + ' — Automated past-due reminder sent');
      if (asOf - o.due > 45) notes.push(fmt(o.due + 30) + ' — Second notice emailed to billing contact');
    }
    if (o.payBehavior === 'partial') notes.push(fmt(o.due + 10) + ' — Short-pay noticed; asked customer AP for remittance detail');
    o.actions = notes.join('\n');
  });

  /* ---------- deliberate test case: two customers with the same name */
  // A second company with exactly the same name as an existing customer, but
  // its own Customer ID, region, billing contact and contract prices. One
  // clean, paid order (and its payment) is moved onto the new ID at the new
  // contract price, so every lookup in the workbook has to use Customer ID.
  // Matching on the name would be ambiguous. Added after the random
  // generation so no other number in the dataset changes.
  let original = null, mover = null;
  for (const c of customers) {
    mover = orders.find(o => o.cust_id === c.id && o.scenario === 'normal' && o.paid !== '' && o.payBehavior === 'ontime');
    if (mover) { original = c; break; }
  }
  const twin = Object.assign({}, original, {
    id: 'C-' + (101 + customers.length), region: 'Gulf Coast', csm: CSMS[(CSMS.findIndex(x => x[0] === original.csm) + 1) % CSMS.length][0],
    csmEmail: CSMS[(CSMS.findIndex(x => x[0] === original.csm) + 1) % CSMS.length][1],
    billName: 'Rowan Haddad', billEmail: 'ap.rowan@' + original.billEmail.split('@')[1].replace('.example', '-gulf.example'), skus: {},
  });
  Object.keys(original.skus).forEach(sku => { twin.skus[sku] = Math.round(original.skus[sku] * 0.93); });
  customers.push(twin);
  Object.keys(twin.skus).forEach(sku => prices.push([twin.id, twin.name, sku, productName[sku], listPrice[sku], twin.skus[sku]]));
  mover.cust = twin; mover.cust_id = twin.id; mover.bill_name = twin.billName; mover.bill_email = twin.billEmail;
  mover.unit_price = twin.skus[mover.sku]; mover.value = mover.qty * mover.unit_price;
  const moverPayment = payments.find(p => p.ref === mover.invoice_no);
  moverPayment.cust_id = twin.id; moverPayment.amount = mover.value;

  /* ---------- ground truth for tests and the write-up */
  const truth = {
    sameName: { name: twin.name, ids: [original.id, twin.id], movedOrder: mover.order_id, invoice: mover.invoice_no, payment: moverPayment.pay_id,
      originalPrice: original.skus[mover.sku], twinPrice: twin.skus[mover.sku], sku: mover.sku },
    asOf,
    holdOrders: orders.filter(o => o.scenario === 'hold').map(o => o.order_id),
    issuesByType: holdIssues.flat().reduce((m, x) => { m[x] = (m[x] || 0) + 1; return m; }, {}),
    stalled: orders.filter(o => o.scenario === 'stalled').map(o => o.order_id),
    slaBreach: orders.filter(o => o.scenario.startsWith('sla_')).map(o => o.order_id),
    duplicates: invoiced.filter(o => o.payBehavior === 'duplicate').map(o => o.invoice_no),
    overpay: invoiced.filter(o => o.payBehavior === 'overpay').map(o => o.invoice_no),
    misref: invoiced.filter(o => o.payBehavior === 'misref').map(o => o.invoice_no),
    partial: invoiced.filter(o => o.payBehavior === 'partial' && payments.some(p => p.ref === o.invoice_no)).map(o => o.invoice_no),
    nonpay: nonPay.map(o => o.invoice_no),
    mysteryPayments: 2,
  };

  return { customers, prices, orders, payments, truth };
}

function serialToIso(serial) {
  const dt = new Date(Math.round((serial - 25569) * 86400000));
  return dt.toISOString().slice(0, 10);
}

function resetSpreadsheet(ss) {
  ss.getNamedRanges().forEach(n => n.remove());

function styleHeader(range, bg) {
  range.setBackground(bg || COLOR.header).setFontColor(COLOR.headerText).setFontWeight('bold')

function writeTable(sheet, headers, rows, widths, headerColor) {
  sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
  styleHeader(sheet.getRange(1, 1, 1, headers.length), headerColor);
  if (rows.length) sheet.getRange(2, 1, rows.length, headers.length).setValues(rows);
  sheet.setFrozenRows(1);
  sheet.setRowHeight(1, 36);
  (widths || []).forEach((w, i) => sheet.setColumnWidth(i + 1, w));
}

function applyColumnFormats(sheet, spec, nRows) {
  spec.forEach((c, i) => {

    sheet.setColumnWidth(i + 1, c.width || 90);
    if (c.note) sheet.getRange(1, i + 1).setNote(c.note);
  });
}

function band(sheet, nRows, nCols) {

}

function orderRules(sh, n) {
  const col = k => sh.getRange(2, OC.I[k], n, 1);

  return [
    R().whenTextEqualTo(HOLD).setBackground(COLOR.red).setFontColor(COLOR.redText).setBold(true).setRanges([col('val_status')]).build(),
    R().whenTextEqualTo(PASS).setFontColor(COLOR.greenText).setRanges([col('val_status')]).build(),
    R().whenCellNotEmpty().setBackground(COLOR.amber).setFontColor(COLOR.amberText).setRanges([col('val_issues')]).build(),
    R().whenCellNotEmpty().setBackground(COLOR.amber).setFontColor(COLOR.amberText).setBold(true).setRanges([col('activation_watch')]).build(),

    R().whenTextEqualTo('61–90').setBackground(COLOR.red).setFontColor(COLOR.redText).setBold(true).setRanges([col('aging')]).build(),
    R().whenTextEqualTo('31–60').setBackground(COLOR.amber).setFontColor(COLOR.amberText).setRanges([col('aging')]).build(),
    R().whenNumberGreaterThan(0).setFontColor(COLOR.redText).setBold(true).setRanges([col('rar')]).build(),

function toSerial(v, tz) {
  if (v === '' || v === null || v === undefined) return '';
  if (typeof v === 'number') return v;
  if (Object.prototype.toString.call(v) === '[object Date]') {
    const iso = Utilities.formatDate(v, tz || 'UTC', 'yyyy-MM-dd');
    return Date.UTC(+iso.slice(0, 4), +iso.slice(5, 7) - 1, +iso.slice(8, 10)) / 86400000 + 25569;
  }
  return v;
}

const ORDER_DATE_KEYS = ['received', 'validated', 'acct_setup', 'shipped', 'activated', 'invoiced', 'paid', 'due_date', 'stage_date'];

function readOrders(ss) {
  const sh = ss.getSheetByName(SHEET.ORDERS);
  const last = sh.getLastRow();
  if (last < 2) return [];
  const tz = ss.getSpreadsheetTimeZone();
  const values = sh.getRange(2, 1, last - 1, ORDER_COLS.length).getValues();
  return values.map((row, i) => {
    const o = { row: i + 2 };
    ORDER_COLS.forEach((c, j) => { o[c.key] = row[j]; });
    ORDER_DATE_KEYS.forEach(k => { o[k] = toSerial(o[k], tz); });
    return o;
  }).filter(o => o.order_id !== '');
}

function readSettings(ss) {
  const v = name => { const r = ss.getRangeByName(name); return r ? r.getValue() : ''; };
  const customers = {};
  const cs = ss.getSheetByName(SHEET.CUSTOMERS);
  if (cs && cs.getLastRow() > 1) {

      .forEach(r => { customers[r[0]] = { name: r[1], csm: r[4], csmEmail: r[5], terms: r[6] }; });

    escDays: Number(v('ESC_DAYS')) || 2,
    collectionsDays: Number(v('COLLECTIONS_DAYS')) || 60,
    customers,
  };
}

/* ------------------------------------------------------------- helpers */

function serialToDate(serial) { return new Date(Math.round((serial - 25569) * 86400000)); }

function fmtDate(serial) {
  if (serial === '' || serial === null) return '';
  const d = serialToDate(serial);
  const M = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
  return M[d.getUTCMonth()] + ' ' + d.getUTCDate() + ', ' + d.getUTCFullYear();
}

function fmtDay(serial) {
  return ['Sun', 'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat'][serialToDate(serial).getUTCDay()] + ' ' + fmtDate(serial);
}

function addBusinessDays(serial, n) {
  let s = serial;
  while (n > 0) {
    s += 1;
    const dow = serialToDate(s).getUTCDay();
    if (dow !== 0 && dow !== 6) n--;
  }
  return s;
}

function plural(n, word) { return n + ' ' + word + (n === 1 ? '' : 's'); }

function money(n) {
  const v = Math.round(Number(n) || 0);
  return (v < 0 ? '-$' : '$') + Math.abs(v).toString().replace(/\B(?=(\d{3})+(?!\d))/g, ',');
}

// {invoice #: [payments]} for unapplied payments the Payments tab thinks belong to that invoice.
function likelyPaymentsByInvoice(payments) {
  const m = {};
  payments.forEach(p => {
    const hit = /^Likely match: (\S+)/.exec(String(p.recon_action || ''));
    if (hit) (m[hit[1]] = m[hit[1]] || []).push(p);
  });
  return m;
}

/**
 * Builds the escalation for one order. Pure function of its inputs so it can
 * be unit-tested outside Sheets.
 * @param {Object} likely output of likelyPaymentsByInvoice(): cash already
 *     received that probably pays this invoice — checked before chasing.
 */
function buildEscalation(o, st, likely) {
  const reasons = String(o.risk_reasons || '');
  if (!reasons) return null;
  const cust = st.customers[o.cust_id] || {};
  const csm = o.csm || cust.csm || 'Account CSM';
  const due = fmtDay(addBusinessDays(st.asOf, st.escDays));
  const what = `${o.qty} × ${o.product}`;
  const problems = [], asks = [];
  const has = s => reasons.indexOf(s) !== -1;

  if (has('Intake hold')) {
    problems.push(`Order ${o.order_id} (${what}, ${money(o.order_value)}) is on intake hold and cannot be entered into the system: ${o.val_issues}. ` +
      `Received ${fmtDate(o.received)} — ${plural(st.asOf - o.received, 'day')} ago.`);
    const issues = String(o.val_issues);
    if (issues.indexOf('Price mismatch') !== -1)
      asks.push(`Account rep / Sales Ops: confirm the order should be re-issued at the contract price of ${money(o.contract_price)}/unit (quoted ${money(o.unit_price)}), or attach an approved pricing exception — by ${due}.`);
    if (issues.indexOf('not on contract') !== -1)
      asks.push(`Sales Ops: confirm the correct SKU, or amend the contract to add ${o.sku} — by ${due}.`);
    if (issues.indexOf('Unknown SKU') !== -1)
      asks.push(`Account rep: correct SKU "${o.sku}" (not in the product catalog) — by ${due}.`);
    if (issues.indexOf('billing contact') !== -1)
      asks.push(`${csm}: get a valid billing contact name and email from the customer so the account can be invoiced — by ${due}.`);
    if (issues.indexOf('≠ contract') !== -1)
      asks.push(`Deal Desk: approve ${o.terms} as an exception or correct the order to contract terms (${o.contract_terms}) — by ${due}.`);
  }
  if (has('Stalled activation')) {
    problems.push(`Order ${o.order_id} (${what}) shipped ${fmtDate(o.shipped)} and has not activated after ${st.asOf - o.shipped} days ` +
      `(threshold ${st.stallDays}). It cannot be invoiced until the customer activates.`);
    asks.push(`${csm}: reach the customer's site lead by ${due}, confirm an install/activation date, and open an RMA if the hardware is faulty. Reply with the committed activation date.`);
  }
  if (has('SLA breach')) {
    const d = o.days_in_stage, sla = o.sla;
    if (o.stage === 'Validated') {
      problems.push(`Order ${o.order_id} was validated ${d} days ago but the account is still not set up (SLA ${sla} days). Shipping is blocked.`);
      asks.push(`Billing Ops: complete account set-up for ${o.customer} by ${due} so the order can ship.`);
    } else if (o.stage === 'Account Set Up') {
      problems.push(`Order ${o.order_id} has been ready to ship for ${d} days (SLA ${sla} days) without a ship date.`);
      asks.push(`Fulfillment: confirm a ship date for ${what} by ${due}; if backordered, give ${csm} a revised ETA to share with the customer.`);
    } else if (o.stage === 'Activated') {
      problems.push(`Order ${o.order_id} activated ${fmtDate(o.activated)} — ${d} days ago — and has not been invoiced (SLA ${sla} days). This is unbilled revenue.`);
      asks.push(`Billing Ops: issue the ${money(o.order_value)} invoice by ${due} and confirm the invoice number.`);
    } else if (o.stage !== 'Received' || !has('Intake hold')) {
      problems.push(`Order ${o.order_id} has been in "${o.stage}" for ${d} days (SLA ${sla} days).`);
      asks.push(`Owner of the ${o.stage} step: unblock by ${due}.`);
    }
  }
  if (has('Past due')) {
    const partial = Number(o.amt_paid) > 0;
    const onFile = (likely && likely[o.invoice_no]) || [];
    problems.push(`Invoice ${o.invoice_no} (${money(o.order_value)}, ${o.terms}) was due ${fmtDate(o.due_date)}. ` +
      `${money(o.balance)} is ${o.dpd} days past due` + (partial ? ` after a partial payment of ${money(o.amt_paid)}.` : '.'));
    if (onFile.length) {
      const p = onFile[0];
      problems.push(`⚠ Likely already paid: payment ${p.pay_id} (${money(p.amount)}, received ${fmtDate(toSerial(p.pay_date))}, remittance ref "${p.remit_ref || 'blank'}") ` +
        `is sitting in unapplied cash and matches this customer and amount.`);
      asks.push(`Cash Applications: verify ${p.pay_id} belongs to ${o.invoice_no} and re-apply it by ${due}. Do NOT send a collections notice to ${o.customer} — they appear to have paid.`);
    } else if (partial) {
      asks.push(`${csm}: find out why ${o.customer} short-paid by ${money(o.balance)} (dispute vs. error) and get a payment date for the remainder — by ${due}.`);
    } else if (Number(o.dpd) > st.collectionsDays) {
      // Decision: the success manager calls first; Collections only this late.
      asks.push(`Collections + ${csm}: secure a written payment commitment for ${money(o.balance)} by ${due}. If none, recommend a credit hold on new orders for ${o.customer}.`);
    } else {
      asks.push(`${csm}: contact ${o.bill_name || 'the billing contact'} (${o.bill_email || 'no email on file'}) and confirm a payment date for ${money(o.balance)} — by ${due}.`);
    }
  }

  const rar = Number(o.rar) || 0;
  const paidButUnapplied = has('Past due') && likely && likely[o.invoice_no];
  const basis = o.invoice_no ? 'open invoice balance' : 'order value not yet invoiceable';
  const actions = String(o.actions || '').trim();
  const actionLines = actions
    ? actions.split('\n').map(a => '• ' + a.trim())
    : ['• None logged yet. Log at least one direct attempt (date — action) in the Orders tab before sending.'];
  // When matching cash is already on file, the subject must not say the money
  // is at risk: the ask is to re-apply the payment, not to chase the customer.
  const subject = paidButUnapplied
    ? `[Escalation] ${o.order_id} · ${o.customer} · Payment likely on file, re-apply ${money(likely[o.invoice_no][0].amount)}`
    : `[Escalation] ${o.order_id} · ${o.customer} · ${money(rar)} at risk — ${reasons.split(';')[0].trim()}`;
  const text = [
    `To: ${csm}${cust.csmEmail ? ' <' + cust.csmEmail + '>' : ''}`,
    st.escCc ? `Cc: ${st.escCc}` : '',
    `Subject: ${subject}`,
    '',
    'PROBLEM',
    problems.join('\n'),
    '',
    'REVENUE AT RISK',
    `${money(rar)} (${basis})`,
    '',
    'ACTIONS ALREADY TAKEN',
    actionLines.join('\n'),
    '',
    'SPECIFIC ASK',
    asks.map(a => '• ' + a).join('\n'),
  ].filter((l, i) => !(i === 1 && l === '')).join('\n');

  return {
    order_id: o.order_id, customer: o.customer, csm, csmEmail: cust.csmEmail || '', rar,
    subject, problem: problems.join('\n'), actions: actionLines.join('\n'), ask: asks.map(a => '• ' + a).join('\n'), text,
  };
}

function mailtoFormula(draft, cc) {
  const body = draft.text.split('\n').slice(draft.text.indexOf('Cc:') === -1 ? 3 : 4).join('\n');

    '?subject=' + encodeURIComponent(draft.subject) +
    (cc ? '&cc=' + encodeURIComponent(cc) : '') +
    '&body=' + encodeURIComponent(body);
  return `=HYPERLINK("${url}","✉ Open email")`;
}

function draftEscalationForRow(ss, row) {
  SpreadsheetApp.flush();
  const st = readSettings(ss);
  const o = readOrders(ss).find(x => x.row === row);
  if (!o) return { error: 'Row ' + row + ' is not an order.' };
  const draft = buildEscalation(o, st, likelyPaymentsByInvoice(readPayments(ss)));
  if (!draft) return { error: `${o.order_id} has no risk flags — nothing to escalate.` };
  const now = new Date();
  appendEscalation(ss, draft, st, now);
  ss.getSheetByName(SHEET.ORDERS).getRange(row, OC.I.last_escalation).setValue(now);
  return { draft };
}

  const kinds = ['Stalled activation', 'Past due', 'Past due', 'Intake hold', 'SLA breach: Activated', 'SLA breach: Validated', 'SLA breach: Account Set Up'];

  });

}

/**
 * Computes the digest. Pure apart from reading the sheet, so it can be
 * previewed without sending or touching the "new errors" snapshot.
 * @param {string[]|null} previousHoldIds order IDs on hold at the last digest
 */
function computeDigest(orders, payments, st, previousHoldIds) {
  const holds = orders.filter(o => o.val_status === HOLD);
  const prev = previousHoldIds ? new Set(previousHoldIds) : null;
  const newHolds = prev ? holds.filter(o => !prev.has(o.order_id)) : holds;
  const stalled = orders.filter(o => String(o.activation_watch) !== '').sort((a, b) => b.rar - a.rar);
  const pastDue = orders.filter(o => Number(o.dpd) > 0);
  const buckets = AGING_BUCKETS.slice(1).map(b => {
    const rows = pastDue.filter(o => o.aging === b);
    return { bucket: b, count: rows.length, amount: rows.reduce((s, o) => s + Number(o.balance), 0) };
  });
  const unapplied = payments.filter(p => Number(p.unapplied) > 0.005);
  const slaOther = orders.filter(o => o.sla_breach === BREACH && o.val_status !== HOLD && !o.activation_watch);
  const sum = (arr, k) => arr.reduce((s, x) => s + (Number(x[k]) || 0), 0);
  const likely = likelyPaymentsByInvoice(payments);
  pastDue.forEach(o => { o.note = likely[o.invoice_no] ? 'Payment likely on file: ' + likely[o.invoice_no][0].pay_id + ' — re-apply, don’t chase' : ''; });
  return {
    asOf: st.asOf,
    holds, newHolds, stalled, pastDue, buckets, unapplied, slaOther,
    pastDueTotal: sum(pastDue, 'balance'),
    unappliedTotal: sum(unapplied, 'unapplied'),
    rarTotal: sum(orders, 'rar'),
    topPastDue: pastDue.slice().sort((a, b) => b.balance - a.balance).slice(0, 5),
    // Decision: the digest leads with the money, whatever the cause.
    topRisk: orders.filter(o => Number(o.rar) > 0).sort((a, b) => b.rar - a.rar).slice(0, 5),
  };
}

function renderDigestText(d) {
  return [
    `Order-to-Cash daily digest — as of ${fmtDate(d.asOf)}`,
    `New intake errors: ${d.newHolds.length} | Stalled activations: ${d.stalled.length} | Past-due AR: ${money(d.pastDueTotal)} | Unapplied cash: ${money(d.unappliedTotal)}`,
    `Revenue at risk: ${money(d.rarTotal)}`,
  ].join('\n');
}

function buildDigest(ss) {
  SpreadsheetApp.flush();
  const st = readSettings(ss);
  const raw = PropertiesService.getDocumentProperties().getProperty(DIGEST_SNAPSHOT_KEY);
  const d = computeDigest(readOrders(ss), readPayments(ss), st, raw ? JSON.parse(raw) : null);
  d.stallDays = st.stallDays;
  d.subject = `O2C digest ${fmtDate(st.asOf)}: ${d.newHolds.length} new errors · ${d.stalled.length} stalled · ${money(d.pastDueTotal)} past due`;
  d.html = renderDigestHtml(d);
  d.text = renderDigestText(d);
  d.recipient = st.digestTo;
  return d;
}

function logDigest(ss, d, delivery) {
  const sh = ss.getSheetByName(SHEET.DIGEST);
  const row = sh.getLastRow() + 1;
  sh.getRange(row, 1, 1, DIGEST_HEADERS.length).setValues([[new Date(), d.recipient || '(none set)', d.newHolds.length, d.holds.length,
    d.stalled.length, d.pastDueTotal, d.unappliedTotal, d.rarTotal, delivery]]);
  sh.getRange(row, 1).setNumberFormat('yyyy-mm-dd hh:mm');
  sh.getRange(row, 6, 1, 3).setNumberFormat(FMT.money0);
  sh.getRange(row, 1, 1, DIGEST_HEADERS.length).setFontSize(9);
}

// Time-driven trigger target + menu: O2C Tools ▸ Automation ▸ Send digest now
function sendDailyDigest() {
  const ss = SpreadsheetApp.getActive();
  const d = buildDigest(ss);
  let delivery;
  if (!d.recipient) {
    delivery = 'Not sent — no recipient in Settings';
  } else {
    MailApp.sendEmail({ to: d.recipient, subject: d.subject, body: d.text, htmlBody: d.html, name: 'O2C Tracker' });
    delivery = 'Emailed';
  }
  logDigest(ss, d, delivery);
  PropertiesService.getDocumentProperties().setProperty(DIGEST_SNAPSHOT_KEY, JSON.stringify(d.holds.map(o => o.order_id)));
  return d;
}

/**
 * Simple trigger: ticking "Escalate" on an order drafts the escalation.
 * Runs without authorisation, so it also works in a freshly copied sheet.
 */
function onEdit(e) {
  if (!e || !e.range) return;
  const sh = e.range.getSheet();
  if (sh.getName() !== SHEET.ORDERS || e.range.getColumn() !== OC.I.escalate || e.range.getNumRows() !== 1) return;
  if (String(e.value).toUpperCase() !== 'TRUE') return;
  const ss = e.source;
  const res = draftEscalationForRow(ss, e.range.getRow());
  e.range.setValue(false);
  ss.toast(res.error || `${res.draft.order_id}: ${money(res.draft.rar)} at risk. Draft added to the Escalations tab.`,
    res.error ? 'Nothing to escalate' : 'Escalation drafted', 6);
}

function installTriggers() {
  removeTriggers();
  const hour = readSettings(SpreadsheetApp.getActive()).digestHour;
  ScriptApp.newTrigger('sendDailyDigest').timeBased().everyDays(1).atHour(hour).create();
}

function removeTriggers() {
  ScriptApp.getProjectTriggers()
    .filter(t => t.getHandlerFunction() === 'sendDailyDigest')
    .forEach(t => ScriptApp.deleteTrigger(t));
}

// Appends a blank order with every formula in place; validation runs as you type.
function addOrderRow() {
  const ss = SpreadsheetApp.getActive();
  const sh = ss.getSheetByName(SHEET.ORDERS);
  const r = sh.getLastRow() + 1;
  const ids = sh.getRange(2, 1, r - 2, 1).getValues().map(x => Number(String(x[0]).replace('SO-', '')) || 0);

}
